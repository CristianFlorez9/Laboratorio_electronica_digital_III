150def cambio_estado_tiempo(tiempo_transcurrido):
151 """
152 @brief Controla las transiciones de estado dependientes del tiempo.
153
154 @details Gestiona las pausas entre niveles, tiempos de penalización,
155 y el paso de la fase de presentación a la fase de inputs.
156
157 @param tiempo_transcurrido Milisegundos actuales del sistema.
158 """
159 global estado_actual, nivel_actual, aux_tiempo_presentacion
160 global aux_tiempo_ingreso, vidas, tiempo_nivel_fallido
161
162 # Nivel completado
163 if estado_actual == NIVEL_COMPLETO:
164 if tiempo_transcurrido - tiempo_nivel_completo >= DURACION_VICTORIA:
165 nivel_actual += 1
166 calcular_tiempo_presentacion()
167 generar_siguiente_elemento()
168 aux_tiempo_presentacion = tiempo_transcurrido
169 estado_actual = PRESENTACION
170 return
171
172 # Nivel fallido
173 if estado_actual == NIVEL_FALLIDO:
174 if tiempo_transcurrido - tiempo_nivel_fallido >= DURACION_FALLO:
175 if vidas <= 0:
176 estado_actual = FIN
177 acumular_tiempo(tiempo_transcurrido)
178 else:
179 aux_tiempo_presentacion = tiempo_transcurrido
180 estado_actual = PRESENTACION
181 return
182
183 # Presentación
184 if estado_actual == PRESENTACION:
185 if tiempo_transcurrido - aux_tiempo_presentacion >= tiempo_presentacion_actual:
186 estado_actual = INPUTS
187 aux_tiempo_ingreso = tiempo_transcurrido
188
189 # INPUTs
190 elif estado_actual == INPUTS:
191 if tiempo_transcurrido - aux_tiempo_ingreso >= tiempo_ingreso:
192 acumular_tiempo(tiempo_transcurrido)
193 contador_botones = 0
194 vidas -= 1
195 tiempo_nivel_fallido = tiempo_transcurrido
196 estado_actual = NIVEL_FALLIDO

289def obtener_led_actual_presentacion(tiempo_transcurrido):
290 """
291 @brief Determina qué LED debe estar encendido en la animación de presentación.
292
293 @param tiempo_transcurrido (int) Milisegundos actuales del sistema.
294 @return (int) Índice del LED a encender (0 a 3), o -1 si todos deben estar apagados.
295 """
296 if estado_actual != PRESENTACION:
297 return -1
298
299 transcurrido = tiempo_transcurrido - aux_tiempo_presentacion
300
301 if nivel_actual == 0:
302 return -1
303
304 tiempo_por_elemento = tiempo_presentacion_actual // nivel_actual
305
306 if tiempo_por_elemento == 0:
307 return -1
308
309 indice = transcurrido // tiempo_por_elemento
310
311 if indice >= nivel_actual:
312 return -1
313
314 dentro_del_elemento = transcurrido % tiempo_por_elemento
315 tiempo_encendido = (tiempo_por_elemento * 70) // 100
316
317 if dentro_del_elemento < tiempo_encendido:
318 return secuencia_leds[indice]
319
320 return -1
