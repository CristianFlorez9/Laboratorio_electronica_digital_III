88def leer_boton_juego():
89 """
90 @brief Controla los botones para adivinar la secuencia.
91
92 @details Lee el arreglo de botones, despues de determinar que esta en estado estable,
93 retorna la posición del boton presionado.
94
95 @return valor del indice correspondiente al boton presionado
96 """
97 global estado_estable_botones, lectura_anterior_botones, tiempo_cambio_botones
98 tiempo_actual = time.ticks_ms()
99
100 for i in range(4):
101 lectura_actual = botones[i].value()
102
103 if lectura_actual != lectura_anterior_botones[i]:
104 tiempo_cambio_botones[i] = tiempo_actual
105 lectura_anterior_botones[i] = lectura_actual
106
107 if time.ticks_diff(tiempo_actual, tiempo_cambio_botones[i]) >= TIEMPO_ANTIRREBOTE:
108 if estado_estable_botones[i] != lectura_actual:
109 estado_estable_botones[i] = lectura_actual
110
111 if estado_estable_botones[i] == 0:
112 return i
113 return -1

115def leer_boton_start(tiempo_actual):
116 """
117 @brief Lee y procesa el estado del botón INICIO/RESET con antirrebote.
118
119 @details Una vez detecta el botón presionado, determina si es un comando de
120 RESET (botón presionado por 2 segundos o más) o un comando de INICIO
121 (botón presionado por menos de 2 segundos). Imprime en el monitor serial
122 el tiempo exacto que permaneció presionado.
123
124 @param tiempo_actual (int) Tiempo actual del sistema en milisegundos para calcular la duración.
125
126 @return (int)
127 - 1: Evento de INICIO (pulsación corta).
128 - 2: Evento de RESET (pulsación larga).
129 - 0: No hay ningún evento nuevo.
130 """
131 global estado_estable_start, lectura_anterior_start, tiempo_cambio_start
132 global tiempo_presionado_start, presionado_start, evento_2seg_generado
133
134 lectura_actual = boton_start.value()
135
136 if lectura_actual != lectura_anterior_start:
137 tiempo_cambio_start = tiempo_actual
138 lectura_anterior_start = lectura_actual
139
140 if time.ticks_diff(tiempo_actual, tiempo_cambio_start) >= TIEMPO_ANTIRREBOTE:
141 if estado_estable_start != lectura_actual:
142 estado_estable_start = lectura_actual
143
144 if estado_estable_start == 0: # Botón presionado
145 if not presionado_start:
146 presionado_start = True
147 evento_2seg_generado = False
148 tiempo_presionado_start = tiempo_actual
149 elif not evento_2seg_generado:
150 tiempo_sostenido = time.ticks_diff(tiempo_actual, tiempo_presionado_start)
151 if tiempo_sostenido >= 2000:
152 evento_2seg_generado = True
153 print(f"\n[BOTÓN START] Tiempo presionado: {tiempo_sostenido} ms -> Evento: RESET")
154 return 2
155 else:
156 if presionado_start:
157 presionado_start = False
158 tiempo_sostenido = time.ticks_diff(tiempo_actual, tiempo_presionado_start)
159 if not evento_2seg_generado and tiempo_sostenido < 2000:
160 print(f"\n[BOTÓN START] Tiempo presionado: {tiempo_sostenido} ms -> Evento: INICIO")
161 return 1
162
163 return 0

185def actualizar_palpito(porcentaje_tiempo, tiempo_actual):
186 """
187 @brief Control del led que plapita.
188
189 @details Contorla que tan rápido palpita el led de acuerdo al tiempo restante para
190 que el usuario ingrese la secuencia vista.
191
192 @param porcentaje_tiempo porcentaje del tiempo restante de respuesta.
193 @param tiempo_actual para calcular el timepo de encendido y apagado.
194 """
195 global ultima_conmutacion_palpito, estado_led_palpito
196
197 if porcentaje_tiempo <= 0.0:
198 led_palpito.value(0)
199 return
200
201 if porcentaje_tiempo > 1.0:
202 porcentaje_tiempo = 1.0
203
204 factor = (1.0 - porcentaje_tiempo)
205 intervalo = 40 + int(560.0 * (factor * factor))
206
207 if time.ticks_diff(tiempo_actual, ultima_conmutacion_palpito) >= intervalo:
208 ultima_conmutacion_palpito = tiempo_actual
209 estado_led_palpito = not estado_led_palpito
210 led_palpito.value(1 if estado_led_palpito else 0)
