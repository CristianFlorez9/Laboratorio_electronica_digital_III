66int leer_boton_juego() {
67 unsigned long tiempo_actual = millis();
68
69 for (int i = 0; i < 4; i++) {
70 bool lectura_actual = digitalRead(PINES_BOTONES[i]);
71
72 if (lectura_actual != lectura_anterior_botones[i]) {
73 tiempo_cambio_botones[i] = tiempo_actual;
74 lectura_anterior_botones[i] = lectura_actual;
75 }
76
77 if (tiempo_actual - tiempo_cambio_botones[i] >= TIEMPO_ANTIRREBOTE) {
78 if (estado_estable_botones[i] != lectura_actual) {
79 estado_estable_botones[i] = lectura_actual;
80
81 if (estado_estable_botones[i] == LOW) {
82 return i;
83 }
84 }
85 }
86 }
87
88 return -1;
89}

198void actualizar_palpito(float porcentaje_tiempo, unsigned long tiempo_actual) {
199 if (porcentaje_tiempo <= 0.0f) {
200 digitalWrite(PIN_LED_PALPITO, LOW);
201 return;
202 }
203
204 if (porcentaje_tiempo > 1.0f) {
205 porcentaje_tiempo = 1.0f;
206 }
207
208 float factor = (1.0f - porcentaje_tiempo);
209 int intervalo = 40 + (int)(560.0f * (factor * factor));
210
211 static unsigned long ultima_conmutacion = 0;
212 static bool estado_led = false;
213
214 if (tiempo_actual - ultima_conmutacion >= (unsigned long)intervalo) {
215 ultima_conmutacion = tiempo_actual;
216 estado_led = !estado_led;
217 digitalWrite(PIN_LED_PALPITO, estado_led ? HIGH : LOW);
218 }
219}
