1/**
2 * @file UART.c
3 * @brief Driver de transmisión de UART0 por interrupciones (interfaz en UART.h).
4 *
5 * @details
6 * El buffer de TX es circular, con dos contadores crecientes:
7 * - tx_tail: dónde se escribe. Solo lo avanza uart_driver_send_string().
8 * - tx_head: dónde se lee. Solo lo avanza tx_fill_hw().
9 *
10 * tx_fill_hw() la llaman la ISR y send_string (esta con las interrupciones apagadas),
11 * así que nunca se ejecuta dos veces a la vez. La interrupción de TX solo está
12 * habilitada mientras queden datos en el buffer.
13 */

22static volatile char tx_buffer[TX_BUFFER_SIZE];
23static volatile uint32_t tx_head = 0;
24static volatile uint32_t tx_tail = 0;
25static volatile uint32_t dropped = 0; /* líneas descartadas por falta de espacio */

27/* Pasa a la FIFO de TX lo que quepa del buffer, sin esperar. */
28static void tx_fill_hw(void) {
29 uart_hw_t *u = uart_get_hw(UART_ID);
30 while (tx_head != tx_tail && !(u->fr & UART_UARTFR_TXFF_BITS)) { /* FIFO no llena */
31 u->dr = tx_buffer[tx_head % TX_BUFFER_SIZE];
32 tx_head++;
33 }
34}
35
36/* ISR: la FIFO de TX se vació hasta el umbral; sigue con lo pendiente. */
37static void uart_isr(void) {
38 tx_fill_hw();
39 if (tx_head == tx_tail) {
40 uart_get_hw(UART_ID)->imsc = 0; /* nada pendiente: apagar la interrupción de TX */
41 }
42}

44void uart_driver_init(void) {
45 uart_init(UART_ID, BAUD_RATE); /* deja 8N1 y sin control de flujo */
46 gpio_set_function(UART_TX_PIN, GPIO_FUNC_UART);
47 uart_set_fifo_enabled(UART_ID, true);
48 uart_get_hw(UART_ID)->ifls = 0; /* IRQ de TX cuando la FIFO baja a 1/8 (4 bytes) */
49
50 irq_set_exclusive_handler(UART0_IRQ, uart_isr);
51 irq_set_enabled(UART0_IRQ, true);
52}

54bool uart_driver_send_string(const char *str) {
55 uint32_t len = strlen(str);
56 uint32_t irq_state = save_and_disable_interrupts(); /* la ISR no entra mientras se copia */
57
58 if (len > TX_BUFFER_SIZE - (tx_tail - tx_head)) { /* no cabe entera: descartar */
59 dropped++;
60 restore_interrupts(irq_state);
61 return false;
62 }
63
64 for (uint32_t i = 0; i < len; i++) {
65 tx_buffer[(tx_tail + i) % TX_BUFFER_SIZE] = str[i];
66 }
67 tx_tail += len;
68
69 tx_fill_hw(); /* a la FIFO lo que quepa ya */
70 if (tx_head != tx_tail) {
71 uart_get_hw(UART_ID)->imsc = UART_UARTIMSC_TXIM_BITS; /* el resto lo envía la ISR */
72 }
73
74 restore_interrupts(irq_state);
75 return true;
76}
