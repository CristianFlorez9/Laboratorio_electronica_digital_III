44/* Un flanco: avanza el emparejamiento y acumula si se completó un periodo. */
45static void handle_edge(const edge_event_t *e) {
46 last_edge_t = e->t_us;
47
48 if (e->rising) {
49 if (estado == VI_BAJADA) { /* periodo completo */
50 sum_T += (uint32_t)(e->t_us - t_subida);
51 sum_H += t_alto;
52 n_periods++;
53 }
54 t_subida = e->t_us;
55 estado = VI_SUBIDA;
56 } else if (estado == VI_SUBIDA) {
57 t_alto = e->t_us - t_subida;
58 estado = VI_BAJADA;
59 } else {
60 estado = ESPERO_SUBIDA; /* bajada inesperada: reiniciar */
61 }
62}

64void measure_poll(void) {
65 /* 1. Procesar los flancos pendientes (si se perdieron, descartar la ventana). */
66 uint32_t ov = capture_overflow_count();
67 if (ov != seen_overflow) {
68 seen_overflow = ov;
69 estado = ESPERO_SUBIDA;
70 sum_T = sum_H = n_periods = 0;
71 }
72 edge_event_t e;
73 while (capture_pop_edge(&e)) handle_edge(&e);
74
75 /* 2. ¿Toca reporte? */
76 uint32_t cnt, tick_t;
77 capture_get_tick(&cnt, &tick_t);
78 if (cnt == reports_seen) return;
79
80 uint32_t n = cnt - 1;
81 uint32_t dt_ms = reports_seen ? (tick_t - prev_tick_t + 500u) / 1000u : 0;
82 late_reports += cnt - reports_seen - 1;
83 reports_seen = cnt;
84 prev_tick_t = tick_t;
85
86 /* 3. Calcular. */
87 bool timeout = (int32_t)(tick_t - last_edge_t) > NO_TRANSITION_TIMEOUT_US;
88 if (timeout) {
89 have_meas = false;
90 estado = ESPERO_SUBIDA;
91 } else if (n_periods > 0) {
92 last_f = n_periods * 1e6 / sum_T;
93 last_d = 100.0 * sum_H / sum_T;
94 have_meas = true;
95 }
96 sum_T = sum_H = n_periods = 0; /* nueva ventana */
97
98 /* 4. Enviar. */
99 char line[64];
100 if (timeout) snprintf(line, sizeof line, "%lu,%lu,NA,NA,SIN_TRANSICIONES\n",
101 (unsigned long)n, (unsigned long)dt_ms);
102 else if (have_meas) snprintf(line, sizeof line, "%lu,%lu,%.3f,%.1f,OK\n",
103 (unsigned long)n, (unsigned long)dt_ms, last_f, last_d);
104 else snprintf(line, sizeof line, "%lu,%lu,NA,NA,ESPERANDO\n",
105 (unsigned long)n, (unsigned long)dt_ms);
106 uart_driver_send_string(line);
107}
