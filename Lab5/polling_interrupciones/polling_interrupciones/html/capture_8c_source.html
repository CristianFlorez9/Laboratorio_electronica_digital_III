28/* ISR de PWM: una vez por flanco. Guarda el instante y el tipo, sin calcular. */
29static void pwm_isr(void) {
30 uint32_t t = time_us_32();
31 uint32_t pending = pwm_get_irq_status_mask(); /* ambos slices comparten la IRQ */
32
33 for (int i = 0; i < 2; i++) {
34 uint slice = i ? fall_slice : rise_slice;
35 if (!(pending & (1u << slice))) continue;
36 pwm_clear_irq(slice);
37
38 if (w_idx - r_idx >= EDGE_RING_SIZE) { /* anillo lleno: descartar y contar */
39 overflow++;
40 } else {
41 ring[w_idx % EDGE_RING_SIZE].t_us = t;
42 ring[w_idx % EDGE_RING_SIZE].rising = (i == 0);
43 w_idx++;
44 }
45 }
46}

56/* Slice que cuenta flancos del pin B; TOP = 0 para un wrap (y una IRQ) por flanco. */
57static uint counter_init(uint gpio, enum pwm_clkdiv_mode mode) {
58 gpio_set_function(gpio, GPIO_FUNC_PWM);
59 uint slice = pwm_gpio_to_slice_num(gpio);
60
61 pwm_config cfg = pwm_get_default_config();
62 pwm_config_set_clkdiv_mode(&cfg, mode);
63 pwm_config_set_wrap(&cfg, 0);
64 pwm_init(slice, &cfg, false); /* queda detenido */
65
66 pwm_clear_irq(slice);
67 pwm_set_irq_enabled(slice, true);
68 return slice;
69}

71void capture_init(void) {
72 rise_slice = counter_init(CAPTURE_RISE_GPIO, PWM_DIV_B_RISING);
73 fall_slice = counter_init(CAPTURE_FALL_GPIO, PWM_DIV_B_FALLING);
74
75 irq_set_exclusive_handler(PWM_IRQ_WRAP, pwm_isr);
76 irq_set_priority(PWM_IRQ_WRAP, 0x00); /* la más alta, para no perder flancos */
77 irq_set_enabled(PWM_IRQ_WRAP, true);
78 pwm_set_mask_enabled((1u << rise_slice) | (1u << fall_slice));
79
80 /* Retardo negativo: periodo fijo entre inicios, sin deriva. */
81 add_repeating_timer_us(-(int32_t)REPORT_PERIOD_US, tick_isr, NULL, &report_timer);
82}
