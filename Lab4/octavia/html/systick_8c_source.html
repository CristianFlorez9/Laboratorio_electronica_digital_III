54void systick_init(uint32_t reload_value)
55{
56 *SYST_CSR = 0; /* apagar mientras se configura */
57
58 /* Prioridad del SysTick: byte alto de SHPR3 (bits 31:24). */
59 *SCB_SHPR3 = (*SCB_SHPR3 & 0x00FFFFFFu) | ((uint32_t)SYSTICK_PRIORIDAD << 24);
60
61 *SYST_RVR = reload_value & SYST_MASCARA_24;
62 *SYST_CVR = 0; /* cualquier escritura limpia el contador */
63
64 *SYST_CSR = SYST_CSR_CLKSOURCE | SYST_CSR_TICKINT | SYST_CSR_ENABLE;
65}
