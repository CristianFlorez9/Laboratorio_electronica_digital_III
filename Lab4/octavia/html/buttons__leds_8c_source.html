127void bl_iniciar(void)
128{
129 for (uint8_t p = 0; p < NUM_BANK0_GPIOS; p++) {
130 indice_de_pin[p] = BL_SIN_BOTON;
131 }
132
133 for (uint8_t i = 0; i < BL_NUM_OCTAVAS; i++) {
134 gpio_init(pines_leds[i]);
135 gpio_set_dir(pines_leds[i], GPIO_OUT);
136 bl_led_escribir(i, false);
137 }
138
139 estado_estable = 0;
140 pendientes = 0;
141 eventos_presionado = 0;
142 eventos_liberado = 0;
143 octava = 0;
144
145 for (uint8_t i = 0; i < BL_NUM_BOTONES; i++) {
146 indice_de_pin[pines_botones[i]] = i;
147 t_flanco_us[i] = 0;
148 t_evento_us[i] = 0;
149
150 gpio_init(pines_botones[i]);
151 gpio_set_dir(pines_botones[i], GPIO_IN);
152#if BL_BOTON_ACTIVO_BAJO
153 gpio_pull_up(pines_botones[i]);
154#else
155 gpio_pull_down(pines_botones[i]);
156#endif
157 }
158
159 sleep_us(100);
160
161 for (uint8_t i = 0; i < BL_NUM_BOTONES; i++) {
162 if (leer_boton_crudo(i)) {
163 estado_estable |= (uint16_t)(1u << i);
164 }
165 }
166
167#if BL_LEDS_SIGUEN_OCTAVA
168 leds_mostrar_octava();
169#endif
170
171 gpio_set_irq_callback(&gpio_isr);
172 for (uint8_t i = 0; i < BL_NUM_BOTONES; i++) {
173 gpio_acknowledge_irq(pines_botones[i], BL_EVENTOS_GPIO);
174 gpio_set_irq_enabled(pines_botones[i], BL_EVENTOS_GPIO, true);
175 }
176 irq_set_enabled(IO_IRQ_BANK0, true);
177}

179void bl_actualizar(void)
180{
181 uint16_t pend = pendientes;
182 if (pend == 0) {
183 return;
184 }
185
186 for (uint8_t i = 0; i < BL_NUM_BOTONES; i++) {
187 if (!((pend >> i) & 1u)) {
188 continue;
189 }
190 if ((uint32_t)(time_us_32() - t_flanco_us[i]) < BL_ANTIRREBOTE_US) {
191 continue;
192 }
193
194 /* La ISR también escribe pendientes: la limpieza debe ser atómica. */
195 uint32_t irq = save_and_disable_interrupts();
196 pendientes &= (uint16_t)~(1u << i);
197 restore_interrupts(irq);
198
199 bool crudo = leer_boton_crudo(i);
200 bool estable = (estado_estable >> i) & 1u;
201
202 if (crudo != estable) {
203 estado_estable ^= (uint16_t)(1u << i);
204 t_evento_us[i] = t_flanco_us[i];
205 if (crudo) {
206 eventos_presionado |= (uint16_t)(1u << i);
207 if (i >= BL_NUM_NOTAS) {
208 octava = (uint8_t)(i - BL_NUM_NOTAS);
209#if BL_LEDS_SIGUEN_OCTAVA
210 leds_mostrar_octava();
211#endif
212 }
213 } else {
214 eventos_liberado |= (uint16_t)(1u << i);
215 }
216 }
217
218 /* Borra el flanco latente del rebote y rehabilita el pin. */
219 gpio_acknowledge_irq(pines_botones[i], BL_EVENTOS_GPIO);
220 gpio_set_irq_enabled(pines_botones[i], BL_EVENTOS_GPIO, true);
221
222 /* Si el nivel cambió justo antes de rehabilitar, reiniciar el filtro. */
223 if (leer_boton_crudo(i) != (bool)((estado_estable >> i) & 1u)) {
224 gpio_set_irq_enabled(pines_botones[i], BL_EVENTOS_GPIO, false);
225 t_flanco_us[i] = time_us_32();
226
227 irq = save_and_disable_interrupts();
228 pendientes |= (uint16_t)(1u << i);
229 restore_interrupts(irq);
230 }
231 }
232}

239void bl_notas_tomar_eventos(uint8_t *presionadas, uint8_t *liberadas)
240{
241 const uint16_t mascara = (1u << BL_NUM_NOTAS) - 1u;
242
243 uint32_t irq = save_and_disable_interrupts();
244 uint16_t p = eventos_presionado & mascara;
245 uint16_t l = eventos_liberado & mascara;
246 eventos_presionado &= (uint16_t)~mascara;
247 eventos_liberado &= (uint16_t)~mascara;
248 restore_interrupts(irq);
249
250 if (presionadas) {
251 *presionadas = (uint8_t)p;
252 }
253 if (liberadas) {
254 *liberadas = (uint8_t)l;
255 }
